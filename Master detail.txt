
CREATE TABLE DMS_Master
(
    MasterID INT IDENTITY(1,1) PRIMARY KEY,
    DocumentNo VARCHAR(50),
    ReferenceNo VARCHAR(50),
    Subject VARCHAR(250),
    DocumentDate DATE,
    Department VARCHAR(150),
    CreatedBy VARCHAR(100),
    CreatedDate DATETIME DEFAULT GETDATE(),
    Status VARCHAR(50),
    Remarks VARCHAR(500)
);


CREATE TABLE DMS_Detail
(
    DetailID INT IDENTITY(1,1) PRIMARY KEY,
    MasterID INT NOT NULL,
    DetailType VARCHAR(100),
    Description VARCHAR(500),
    ReferenceNo VARCHAR(100),
    Remarks VARCHAR(500),

    CONSTRAINT FK_DMS_Detail_Master
    FOREIGN KEY (MasterID)
    REFERENCES DMS_Master(MasterID)
);

var masterTable;
var detailTable;
var selectedMaster = null;
var detailRequest = null;


// INITIALIZE PAGE

$(document).ready(function () {


    // MASTER DATATABLE

    masterTable = $('#tblMaster').DataTable({

        pageLength: 10,
        ordering: true,
        searching: true,
        autoWidth: false,

        columns: [

            { data: 'DocumentNo' },
            { data: 'ReferenceNo' },
            { data: 'Subject' },
            { data: 'Department' },
            { data: 'DocumentDate' },

            {
                data: 'Status',
                render: function (data) {

                    var status = (data || '').toLowerCase();
                    var cls = 'label-info';

                    if (status === 'approved')
                        cls = 'label-success';

                    else if (status === 'pending')
                        cls = 'label-warning';

                    else if (status === 'rejected')
                        cls = 'label-danger';

                    return '<span class="label ' + cls + '">' +
                        escapeHtml(data) + '</span>';
                }
            },

            {
                data: 'MasterID',
                orderable: false,
                searchable: false,

                render: function (id) {

                    return '<button type="button" ' +
                        'class="btn btn-info btn-xs btnViewMaster" ' +
                        'data-id="' + id + '">' +
                        '<i class="fa fa-eye"></i> View' +
                        '</button>';
                }
            }

        ]

    });


    // DETAIL DATATABLE

    detailTable = $('#tblDetails').DataTable({

        pageLength: 10,
        searching: true,
        ordering: true,
        autoWidth: false,

        columns: [

            { data: null },

            { data: 'DetailType' },

            { data: 'Description' },

            { data: 'ReferenceNo' },

            { data: 'Remarks' },

            {
                data: 'DetailID',
                orderable: false,
                searchable: false,

                render: function (id) {

                    return '<button type="button" ' +
                        'class="btn btn-primary btn-xs btnEditDetail" ' +
                        'data-id="' + id + '">' +
                        '<i class="fa fa-edit"></i> Edit' +
                        '</button>';
                }
            }

        ],

        columnDefs: [{
            targets: 0,
            render: function (data, type, row, meta) {
                return meta.row +
                    meta.settings._iDisplayStart + 1;
            }
        }]

    });


    // LOAD MASTER RECORDS

    loadMasters();


    // VIEW MASTER BUTTON

    $('#tblMaster tbody').on('click',
        '.btnViewMaster', function (e) {

        e.stopPropagation();

        var row = masterTable.row(
            $(this).closest('tr')
        ).data();

        if (!row) return;

        selectedMaster = row;

        showMasterInfo(row);

        // Clear old detail records

        detailTable.clear().draw();

        // Open master detail modal

        $('#masterDetailModal').modal('show');

        // Load selected master's details

        loadDetails(row.MasterID);

    });


    // ADD DETAIL BUTTON

    $('#btnAddDetail').click(function () {

        if (!selectedMaster) return;

        clearDetailForm();

        $('#hdnMasterID').val(selectedMaster.MasterID);

        $('#detailModalTitle').html(
            '<i class="fa fa-plus"></i> Add Detail'
        );

        $('#detailEditModal').modal('show');

    });


    // EDIT DETAIL BUTTON

    $('#tblDetails tbody').on('click',
        '.btnEditDetail', function () {

        var row = detailTable.row(
            $(this).closest('tr')
        ).data();

        if (!row) return;

        clearDetailForm();

        $('#hdnDetailID').val(row.DetailID);

        $('#hdnMasterID').val(selectedMaster.MasterID);

        $('#ddlDetailType').val(row.DetailType);

        $('#txtDescription').val(row.Description);

        $('#txtReferenceNo').val(row.ReferenceNo);

        $('#txtDetailRemarks').val(row.Remarks);

        $('#detailModalTitle').html(
            '<i class="fa fa-edit"></i> Edit Detail'
        );

        $('#detailEditModal').modal('show');

    });


    // SAVE DETAIL BUTTON

    $('#btnSaveDetail').click(function () {

        var detailId = parseInt(
            $('#hdnDetailID').val() || 0
        );

        var masterId = parseInt(
            $('#hdnMasterID').val() || 0
        );

        var detailType = $('#ddlDetailType').val();

        var description = $.trim(
            $('#txtDescription').val()
        );

        if (!masterId || !detailType || !description) {

            alert('Please fill all required fields.');

            return;
        }


        var data = {

            detailId: detailId,

            masterId: masterId,

            detailType: detailType,

            description: description,

            referenceNo: $('#txtReferenceNo').val(),

            remarks: $('#txtDetailRemarks').val()

        };


        $('#btnSaveDetail')
            .prop('disabled', true)
            .html('<i class="fa fa-spinner fa-spin"></i> Saving...');


        $.ajax({

            type: "POST",

            url: "MasterDetail.aspx/SaveDetail",

            data: JSON.stringify(data),

            contentType: "application/json; charset=utf-8",

            dataType: "json",

            success: function (response) {

                $('#btnSaveDetail')
                    .prop('disabled', false)
                    .html('<i class="fa fa-save"></i> Save Detail');

                if (response.d.Success) {

                    $('#detailEditModal').modal('hide');

                    loadDetails(masterId);

                    alert(response.d.Message);

                }
                else {

                    alert(response.d.Message);

                }

            },

            error: function (xhr) {

                $('#btnSaveDetail')
                    .prop('disabled', false)
                    .html('<i class="fa fa-save"></i> Save Detail');

                console.log(xhr.responseText);

                alert('Unable to save detail record.');

            }

        });

    });


    // EDIT MASTER BUTTON

    $('#btnEditMaster').click(function () {

        if (!selectedMaster) return;

        // Connect your existing master edit modal here.

        alert('Selected Master ID: ' +
            selectedMaster.MasterID);

    });


    // STACKED MODAL BACKDROP

    $('#detailEditModal').on('show.bs.modal', function () {

        $(this).appendTo('body');

    });

    $('#detailEditModal').on('shown.bs.modal', function () {

        $('.modal-backdrop').last().css('z-index', 1055);

        $(this).css('z-index', 1060);

    });

});


// LOAD MASTER RECORDS

function loadMasters() {

    $.ajax({

        type: "POST",

        url: "MasterDetail.aspx/GetMasters",

        data: "{}",

        contentType: "application/json; charset=utf-8",

        dataType: "json",

        success: function (response) {

            masterTable.clear();

            masterTable.rows.add(response.d);

            masterTable.draw();

        },

        error: function (xhr) {

            console.log(xhr.responseText);

            alert('Unable to load master records.');

        }

    });

}


// SHOW MASTER INFORMATION

function showMasterInfo(row) {

    $('#vDocumentNo').text(row.DocumentNo || '-');

    $('#vReference').text(row.ReferenceNo || '-');

    $('#vSubject').text(row.Subject || '-');

    $('#vDocumentDate').text(row.DocumentDate || '-');

    $('#vDepartment').text(row.Department || '-');

    $('#vCreatedBy').text(row.CreatedBy || '-');

    $('#vCreatedDate').text(row.CreatedDate || '-');

    $('#vRemarks').text(row.Remarks || '-');


    var status = row.Status || '-';

    var cls = 'label-info';

    if (status.toLowerCase() === 'approved')
        cls = 'label-success';

    else if (status.toLowerCase() === 'pending')
        cls = 'label-warning';

    else if (status.toLowerCase() === 'rejected')
        cls = 'label-danger';


    $('#vStatus')
        .removeClass('label-info label-success label-warning label-danger')
        .addClass('label ' + cls)
        .text(status);


    $('#lblDetailDocument').text(
        row.DocumentNo + ' - ' + row.Subject
    );

}


// LOAD DETAIL RECORDS

function loadDetails(masterId) {

    if (detailRequest) {
        detailRequest.abort();
    }

    detailTable.clear().draw();

    detailRequest = $.ajax({

        type: "POST",

        url: "MasterDetail.aspx/GetDetails",

        data: JSON.stringify({
            masterId: masterId
        }),

        contentType: "application/json; charset=utf-8",

        dataType: "json",

        success: function (response) {

            if (!selectedMaster ||
                selectedMaster.MasterID != masterId) {
                return;
            }

            detailTable.clear();

            detailTable.rows.add(response.d);

            detailTable.draw();

        },

        error: function (xhr, status) {

            if (status !== 'abort') {

                console.log(xhr.responseText);

                alert('Unable to load detail records.');

            }

        },

        complete: function () {
            detailRequest = null;
        }

    });

}


// CLEAR DETAIL FORM

function clearDetailForm() {

    $('#hdnDetailID').val('');

    $('#hdnMasterID').val('');

    $('#ddlDetailType').val('');

    $('#txtDescription').val('');

    $('#txtReferenceNo').val('');

    $('#txtDetailRemarks').val('');

}


// HTML ENCODING

function escapeHtml(value) {

    return $('<div/>')
        .text(value == null ? '' : value)
        .html();

}

    [WebMethod]
    [ScriptMethod(ResponseFormat = ResponseFormat.Json)]
    public static List<MasterModel> GetMasters()
    {
        List<MasterModel> list = new List<MasterModel>();

        using (SqlConnection con = new SqlConnection(conStr))
        {
            string sql = @"
                SELECT
                    MasterID,
                    DocumentNo,
                    ReferenceNo,
                    Subject,
                    DocumentDate,
                    Department,
                    CreatedBy,
                    CreatedDate,
                    Status,
                    Remarks
                FROM DMS_Master
                ORDER BY MasterID DESC";

            using (SqlCommand cmd = new SqlCommand(sql, con))
            {
                con.Open();

                using (SqlDataReader dr = cmd.ExecuteReader())
                {
                    while (dr.Read())
                    {
                        MasterModel obj = new MasterModel();

                        obj.MasterID = Convert.ToInt32(
                            dr["MasterID"]);

                        obj.DocumentNo = GetString(
                            dr["DocumentNo"]);

                        obj.ReferenceNo = GetString(
                            dr["ReferenceNo"]);

                        obj.Subject = GetString(
                            dr["Subject"]);

                        obj.DocumentDate =
                            dr["DocumentDate"] == DBNull.Value
                            ? ""
                            : Convert.ToDateTime(
                                dr["DocumentDate"]
                              ).ToString("dd-MMM-yyyy");

                        obj.Department = GetString(
                            dr["Department"]);

                        obj.CreatedBy = GetString(
                            dr["CreatedBy"]);

                        obj.CreatedDate =
                            dr["CreatedDate"] == DBNull.Value
                            ? ""
                            : Convert.ToDateTime(
                                dr["CreatedDate"]
                              ).ToString("dd-MMM-yyyy HH:mm");

                        obj.Status = GetString(
                            dr["Status"]);

                        obj.Remarks = GetString(
                            dr["Remarks"]);

                        list.Add(obj);
                    }
                }
            }
        }

        return list;
    }

    // =====================================
    // SAVE DETAIL - INSERT / UPDATE
    // =====================================

    [WebMethod]
    [ScriptMethod(ResponseFormat = ResponseFormat.Json)]
    public static SaveResult SaveDetail(
        int detailId,
        int masterId,
        string detailType,
        string description,
        string referenceNo,
        string remarks)
    {
        SaveResult result = new SaveResult();

        try
        {
            if (masterId <= 0)
            {
                result.Success = false;
                result.Message = "Invalid master record.";
                return result;
            }

            if (string.IsNullOrWhiteSpace(detailType) ||
                string.IsNullOrWhiteSpace(description))
            {
                result.Success = false;
                result.Message = "Required fields are missing.";
                return result;
            }


            using (SqlConnection con = new SqlConnection(conStr))
            {
                con.Open();

                // Verify parent exists

                string checkSql = @"
                    SELECT COUNT(1)
                    FROM DMS_Master
                    WHERE MasterID = @MasterID";

                using (SqlCommand checkCmd =
                    new SqlCommand(checkSql, con))
                {
                    checkCmd.Parameters.Add(
                        "@MasterID",
                        SqlDbType.Int
                    ).Value = masterId;

                    if (Convert.ToInt32(
                        checkCmd.ExecuteScalar()) == 0)
                    {
                        result.Success = false;
                        result.Message = "Master record not found.";
                        return result;
                    }
                }


                if (detailId == 0)
                {
                    // INSERT

                    string insertSql = @"
                        INSERT INTO DMS_Detail
                        (
                            MasterID,
                            DetailType,
                            Description,
                            ReferenceNo,
                            Remarks
                        )
                        VALUES
                        (
                            @MasterID,
                            @DetailType,
                            @Description,
                            @ReferenceNo,
                            @Remarks
                        );

                        SELECT CAST(SCOPE_IDENTITY() AS INT);";


                    using (SqlCommand cmd =
                        new SqlCommand(insertSql, con))
                    {
                        AddDetailParameters(
                            cmd,
                            masterId,
                            detailType,
                            description,
                            referenceNo,
                            remarks
                        );

                        result.ID = Convert.ToInt32(
                            cmd.ExecuteScalar());
                    }

                    result.Success = true;
                    result.Message = "Detail saved successfully.";
                }
                else
                {
                    // UPDATE

                    string updateSql = @"
                        UPDATE DMS_Detail
                        SET
                            DetailType = @DetailType,
                            Description = @Description,
                            ReferenceNo = @ReferenceNo,
                            Remarks = @Remarks
                        WHERE
                            DetailID = @DetailID
                            AND MasterID = @MasterID";


                    using (SqlCommand cmd =
                        new SqlCommand(updateSql, con))
                    {
                        AddDetailParameters(
                            cmd,
                            masterId,
                            detailType,
                            description,
                            referenceNo,
                            remarks
                        );

                        cmd.Parameters.Add(
                            "@DetailID",
                            SqlDbType.Int
                        ).Value = detailId;

                        int affected = cmd.ExecuteNonQuery();

                        if (affected > 0)
                        {
                            result.Success = true;
                            result.Message =
                                "Detail updated successfully.";

                            result.ID = detailId;
                        }
                        else
                        {
                            result.Success = false;
                            result.Message =
                                "Detail record not found.";
                        }
                    }
                }
            }
        }
        catch (Exception ex)
        {
            // Log ex securely on the server.

            result.Success = false;
            result.Message =
                "Unable to save detail record. Please try again.";
        }

        return result;
    }


    // =====================================
    // DETAIL SQL PARAMETERS
    // =====================================

    private static void AddDetailParameters(
        SqlCommand cmd,
        int masterId,
        string detailType,
        string description,
        string referenceNo,
        string remarks)
    {
        cmd.Parameters.Add(
            "@MasterID",
            SqlDbType.Int
        ).Value = masterId;

        cmd.Parameters.Add(
            "@DetailType",
            SqlDbType.VarChar,
            100
        ).Value = detailType.Trim();

        cmd.Parameters.Add(
            "@Description",
            SqlDbType.VarChar,
            500
        ).Value = description.Trim();

        cmd.Parameters.Add(
            "@ReferenceNo",
            SqlDbType.VarChar,
            100
        ).Value = string.IsNullOrWhiteSpace(referenceNo)
            ? (object)DBNull.Value
            : referenceNo.Trim();

        cmd.Parameters.Add(
            "@Remarks",
            SqlDbType.VarChar,
            500
        ).Value = string.IsNullOrWhiteSpace(remarks)
            ? (object)DBNull.Value
            : remarks.Trim();
    }


    // =====================================
    // NULL SAFE STRING
    // =====================================

    private static string GetString(object value)
    {
        return value == DBNull.Value
            ? ""
            : Convert.ToString(value);
    }
