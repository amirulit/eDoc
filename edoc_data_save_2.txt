using System;
using System.Collections.Generic;
using System.Configuration;
using System.Data.SqlClient;
using System.Web.Services;
using System.Web.Script.Services;

public class DropdownItem
{
    public string Value { get; set; }
    public string Text { get; set; }
}

[WebMethod]
[ScriptMethod(ResponseFormat = ResponseFormat.Json)]
public static List<DropdownItem> GetCustomerCodes()
{
    var items = new List<DropdownItem>();

    string cs = ConfigurationManager
        .ConnectionStrings["MyConnection"].ConnectionString;

    using (SqlConnection con = new SqlConnection(cs))
    using (SqlCommand cmd = new SqlCommand(
        @"SELECT CustomerCode
          FROM dbo.Customers
          ORDER BY CustomerCode", con))
    {
        con.Open();

        using (SqlDataReader dr = cmd.ExecuteReader())
        {
            while (dr.Read())
            {
                string code = dr["CustomerCode"].ToString();

                items.Add(new DropdownItem
                {
                    Value = code,
                    Text = code
                });
            }
        }
    }

    return items;
}

[WebMethod]
[ScriptMethod(ResponseFormat = ResponseFormat.Json)]
public static List<DropdownItem> GetLoanNatures()
{
    var items = new List<DropdownItem>();

    string cs = ConfigurationManager
        .ConnectionStrings["MyConnection"].ConnectionString;

    using (SqlConnection con = new SqlConnection(cs))
    using (SqlCommand cmd = new SqlCommand(
        @"SELECT LoanNatureId, LoanNatureName
          FROM dbo.LoanNature
          ORDER BY LoanNatureName", con))
    {
        con.Open();

        using (SqlDataReader dr = cmd.ExecuteReader())
        {
            while (dr.Read())
            {
                items.Add(new DropdownItem
                {
                    Value = dr["LoanNatureId"].ToString(),
                    Text = dr["LoanNatureName"].ToString()
                });
            }
        }
    }

    return items;
}



$(document).ready(function () {

    loadCustomerCodes();
    loadLoanNatures();

    function loadCustomerCodes() {
        $.ajax({
            type: "POST",
            url: "LoanEntry.aspx/GetCustomerCodes",
            data: "{}",
            contentType: "application/json; charset=utf-8",
            dataType: "json",

            success: function (response) {
                var ddl = $("#ddlCustomerCode");
                ddl.empty();
                ddl.append(new Option("-- Select Customer --", ""));

                $.each(response.d, function (i, item) {
                    ddl.append(new Option(item.Text, item.Value));
                });
            },

            error: function (xhr) {
                console.error(xhr.responseText);
                alert("Unable to load customer codes.");
            }
        });
    }

    function loadLoanNatures() {
        $.ajax({
            type: "POST",
            url: "LoanEntry.aspx/GetLoanNatures",
            data: "{}",
            contentType: "application/json; charset=utf-8",
            dataType: "json",

            success: function (response) {
                var ddl = $("#ddlLoanNature");
                ddl.empty();
                ddl.append(new Option("-- Select Loan Nature --", ""));

                $.each(response.d, function (i, item) {
                    ddl.append(new Option(item.Text, item.Value));
                });
            },

            error: function (xhr) {
                console.error(xhr.responseText);
                alert("Unable to load loan natures.");
            }
        });
    }

    // Continue with your existing Add, Remove and Save All code here.
});
